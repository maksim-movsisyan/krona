add_executable(krona_test_convection_core test_operator.cpp fixture.hpp)
target_link_libraries(krona_test_convection_core PRIVATE krona_test_support krona::transport_scalar_convection krona::transport_scalar_diffusion)
if(NOT MPIEXEC_EXECUTABLE)
    message(FATAL_ERROR "Convection core tests require an MPI launcher")
endif()
foreach(case IN ITEMS upwind linear tvd muscl float_storage duplicate empty_rank empty minimal patch_values unlabelled unused_inputs)
    add_test(NAME "transport.scalar.convection.core.${case}" COMMAND "${MPIEXEC_EXECUTABLE}" "${MPIEXEC_NUMPROC_FLAG}" 1 ${MPIEXEC_PREFLAGS} "$<TARGET_FILE:krona_test_convection_core>" ${MPIEXEC_POSTFLAGS} "${case}")
    set_tests_properties("transport.scalar.convection.core.${case}" PROPERTIES TIMEOUT 60 RUN_SERIAL TRUE)
endforeach()
foreach(ranks IN ITEMS 2 3)
    foreach(case IN ITEMS linear tvd patch_values unlabelled unused_inputs empty_rank empty duplicate)
        add_test(NAME "transport.scalar.convection.core.mpi${ranks}.${case}" COMMAND "${MPIEXEC_EXECUTABLE}" "${MPIEXEC_NUMPROC_FLAG}" "${ranks}" ${MPIEXEC_PREFLAGS} "$<TARGET_FILE:krona_test_convection_core>" ${MPIEXEC_POSTFLAGS} "${case}")
        set_tests_properties("transport.scalar.convection.core.mpi${ranks}.${case}" PROPERTIES TIMEOUT 90 PROCESSORS "${ranks}" RUN_SERIAL TRUE)
    endforeach()
endforeach()
set(cases missing_inflow bad_profile null_comm missing_faces)
set(messages "missing or nonfinite inflow value" "missing or nonfinite inflow value" "null communicator" "invalid face geometry or cell-face incidence")
foreach(i RANGE 0 3)
    list(GET cases ${i} case)
    list(GET messages ${i} expected)
    add_test(NAME "transport.scalar.convection.core.reject_${case}" COMMAND "${CMAKE_COMMAND}" "-DPROGRAM=$<TARGET_FILE:krona_test_convection_core>" "-DCASE=${case}" "-DEXPECT=${expected}" "-DRANKS=1" "-DMPIEXEC_EXECUTABLE=${MPIEXEC_EXECUTABLE}" "-DMPIEXEC_NUMPROC_FLAG=${MPIEXEC_NUMPROC_FLAG}" "-DMPIEXEC_PREFLAGS=${MPIEXEC_PREFLAGS}" "-DMPIEXEC_POSTFLAGS=${MPIEXEC_POSTFLAGS}" -P "${CMAKE_CURRENT_SOURCE_DIR}/../../expect_failure.cmake")
    set_tests_properties("transport.scalar.convection.core.reject_${case}" PROPERTIES TIMEOUT 60 RUN_SERIAL TRUE)
endforeach()
